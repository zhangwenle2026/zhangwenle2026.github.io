## allowAgents 修复方案（已准备好）
```
# 需要 root 权限执行：
python3 -c "
import json
with open('/root/.openclaw/openclaw.json') as f:
    d = json.load(f)
agents_list = d.setdefault('agents', {}).setdefault('list', [])
existing_ids = {a['id'] for a in agents_list}
for aid in ['exec', 'ada', 'mentor', 'scout']:
    if aid not in existing_ids:
        agents_list.append({'id': aid, 'workspace': f'/root/.openclaw/workspace-{aid}', 'agentDir': f'/root/.openclaw/agents/{aid}/agent'})
for a in agents_list:
    if a['id'] == 'main':
        a.setdefault('subagents', {})['allowAgents'] = ['exec', 'ada', 'mentor', 'scout']
with open('/root/.openclaw/openclaw.json', 'w') as f:
    json.dump(d, f, indent=2, ensure_ascii=False)
print('done')
"
```
